cmake_minimum_required(VERSION 3.22)
project(HomeKeys VERSION 0.9.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(HOMEKEYS_BUILD_APP "Build the JUCE desktop app" ON)
add_executable(homekeys_tests tests/CoreTests.cpp)
target_include_directories(homekeys_tests PRIVATE src)
enable_testing()
add_test(NAME core COMMAND homekeys_tests)
add_executable(homekeys_travel_tests tests/TravelTests.cpp)
target_include_directories(homekeys_travel_tests PRIVATE src)
add_test(NAME travel COMMAND homekeys_travel_tests)
if(WIN32)
  add_executable(homekeys_analog_tests tests/AnalogTests.cpp)
  target_include_directories(homekeys_analog_tests PRIVATE src)
  target_compile_definitions(homekeys_analog_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(homekeys_analog_tests PRIVATE hid setupapi)
  add_test(NAME analog COMMAND homekeys_analog_tests)
endif()
if(HOMEKEYS_BUILD_APP)
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.9 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
  add_executable(homekeys_lesson_tests tests/LessonTests.cpp)
  target_include_directories(homekeys_lesson_tests PRIVATE src)
  target_compile_definitions(homekeys_lesson_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(homekeys_lesson_tests PRIVATE juce::juce_audio_formats juce::juce_recommended_config_flags)
  if(MSVC)
    target_compile_options(homekeys_lesson_tests PRIVATE /utf-8)
  endif()
  add_test(NAME lesson COMMAND homekeys_lesson_tests)
  add_test(NAME bundled_lessons COMMAND homekeys_lesson_tests "${CMAKE_CURRENT_SOURCE_DIR}/lessons/scale.mid")
  juce_add_gui_app(HomeKeys PRODUCT_NAME "HomeKeys" VERSION "${PROJECT_VERSION}" COMPANY_NAME "HomeKeys" BUNDLE_ID "local.homekeys.desktop" DESCRIPTION "HomeKeys keyboard instrument and MIDI learning")
  if(WIN32)
    # JUCE's generated RC command otherwise retains metadata from the first build.
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/HomeKeys_artefacts/JuceLibraryCode/HomeKeys_resources.rc"
      APPEND DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/HomeKeys_artefacts/JuceLibraryCode/Info.txt")
  endif()
  if(MSVC)
    target_compile_options(HomeKeys PRIVATE /utf-8)
  endif()
  target_sources(HomeKeys PRIVATE src/Main.cpp)
  target_include_directories(HomeKeys PRIVATE src)
  target_compile_definitions(HomeKeys PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(HomeKeys PRIVATE hid setupapi juce::juce_audio_utils juce::juce_recommended_config_flags juce::juce_recommended_lto_flags)
endif()
